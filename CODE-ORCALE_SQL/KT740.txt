SELECT DISTINCT C.CS_MAPGD MAPOS
    , (SELECT P.PO_TEN FROM DMPOS P WHERE P.PO_MA = C.CS_MAPGD) TENPOS
    , SUBSTR(C.CS_MADP, 1, 6) MAXA
    , (SELECT X.TEN FROM DMXA X WHERE X.MA = SUBSTR(C.CS_MADP, 1, 6) AND X.PGD_QL = C.CS_MAPGD) TENXA
    , K.KH_MAKH MAKH
    , K.KH_TENKH TENKH
    , C.CS_SO_TK2 CASA_105
    , K.KH_CMT
    , INITCAP(LOWER(REGEXP_REPLACE(REGEXP_REPLACE(REGEXP_REPLACE(TRIM(K.KH_DIACHI), '(^[-[:space:]]+)|([-[:space:]]+$)', ''), '[-[:space:]]{2,}', '-'), '^[^[:alnum:]]+', ''))) DIACHI
    , K.KH_MOBILE MOBILE
    , C.CS_SODU_TK SODU_TK
    , C.CS_NGAYDKY NGAYDKY
    , NVL(C.CS_MATO, O.TO_MATO) MATO
    , O.TO_TENTT TENTT
FROM CASA_DAILY C
LEFT JOIN HSKH K
    ON C.CS_MAKH = K.KH_MAKH
    AND K.KH_TTRANG = 'A'
LEFT JOIN TMP_HSTO O
    ON C.CS_MATO = O.TO_MATO
    AND O.NG_CAPNHAT = C.CS_NGAYBC
INNER JOIN TRANSPOINT_SUBCOMMUNE_MAPPING M
    ON SUBSTR(C.CS_MADP, 5, 2) = M.XA
    AND SUBSTR(C.CS_MADP, 7, 2) = M.THON
    AND M.NGAY_CAPNHAT = (SELECT MAX(T.NGAY_CAPNHAT) FROM TRANSPOINT_SUBCOMMUNE_MAPPING T)
WHERE C.CS_NGAYBC = TO_DATE(:P_NGAYBC, 'DD/MM/YYYY')
AND C.CS_NGAYDKY BETWEEN TRUNC(TO_DATE(:P_NGAYBC, 'DD/MM/YYYY'), 'MM') AND TO_DATE(:P_NGAYBC, 'DD/MM/YYYY')
AND C.CS_TTSO_TK <> 'C'
AND C.CS_SP_TK = '105'
AND O.TO_LOAITO = '01'
AND SUBSTR(C.CS_MADP, 1, 6) = :P_XA
AND M.MA_DIEM_GDX = :P_DIEMGD
